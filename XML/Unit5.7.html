How the Checklist Works

  1. JavaScript code

    function updateProgress() {

    const boxes = document.querySelectorAll(
        "#checklist input[type='checkbox']"
    );

    let completed = 0;

    boxes.forEach(function (box) {
        if (box.checked) {
            completed++;
        }
    });

    const total = boxes.length;
    const percent = Math.round((completed / total) * 100);

    document.getElementById("progress").textContent =
        completed + " of " + total + " completed";

    document.getElementById("percentage").textContent =
        percent + "%";

    document.getElementById("bar").style.width = percent + "%";
}

function resetChecklist() {

    document.querySelectorAll(
        "#checklist input[type='checkbox']"
    ).forEach(function (box) {
        box.checked = false;
    });

    updateProgress();
}

updateProgress();

  2. Explanation

    updateProgress() counts the ticked boxes, works out the percentage and updates the text and the bar. resetChecklist() unticks every box and then calls updateProgress() again. The last line runs once when the page loads, so the numbers are always correct.
